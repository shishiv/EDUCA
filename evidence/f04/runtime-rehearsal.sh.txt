#!/usr/bin/env bash
set -euo pipefail
cd "$(dirname "$0")/../app"
ROOT=$(cd .. && pwd)
export PATH="$ROOT/.tmp/bin:$PATH"
source scripts/pilot-local-runtime.sh
source scripts/pilot-port-range-lease.sh
source scripts/pilot-supabase-cleanup.sh
export PILOT_E2E_PORT_LEASE_START=62000 PILOT_E2E_PORT_LEASE_END=64000
pilot_port_range_lease_acquire
PROJECT=$(mktemp -d "$ROOT/.tmp/f04-supabase.XXXXXX")
printf '%s\n' "$PROJECT" > "$ROOT/.tmp/stack-path"
APP_PID=''
cleanup() {
  pilot_stop_app_process_group "$APP_PID"
  pilot_supabase_stop_project "$PROJECT" "$(basename "$PROJECT")"
  pilot_port_range_lease_release
}
trap cleanup EXIT
trap 'exit 143' TERM INT
mkdir -p "$PROJECT"
PORT=$((PILOT_E2E_PORT_BASE+9))
pilot_local_project_init "$ROOT" "$PROJECT" "$PILOT_E2E_PORT_BASE" "http://127.0.0.1:$PORT"
pnpm exec supabase --workdir "$PROJECT" start > "$ROOT/.tmp/supabase-start.log" 2>&1
pnpm exec supabase --workdir "$PROJECT" status -o env > "$ROOT/.tmp/local.env"
eval "$(grep -E '^(API_URL|DB_URL|PUBLISHABLE_KEY|SECRET_KEY)=' "$ROOT/.tmp/local.env")"
export NEXT_PUBLIC_SUPABASE_URL="$API_URL" NEXT_PUBLIC_SUPABASE_ANON_KEY="$PUBLISHABLE_KEY" SUPABASE_SERVICE_ROLE_KEY="$SECRET_KEY" SUPABASE_DB_URL="$DB_URL"
export NEXT_PUBLIC_APP_URL="http://127.0.0.1:$PORT"
export NEXT_PUBLIC_PILOT_MODE=true PILOT_MODE=true PILOT_SYNTHETIC_DATA_ONLY=true PILOT_EXTERNAL_DEPLOY_APPROVED=false PILOT_LEGAL_APPROVAL_STATUS=not_approved NEXT_PUBLIC_DEMO_SANDBOX=false DEMO_SANDBOX=false
pnpm exec tsx scripts/pilot-safety-gate.ts seed
pnpm exec supabase --workdir "$PROJECT" db reset --local > "$ROOT/.tmp/reset.log" 2>&1
psql "$DB_URL" -X -v ON_ERROR_STOP=1 -f ../supabase/pilot/provision-pilot-module-gate.sql
pnpm exec tsx scripts/seed-pilot-synthetic.ts
printf '%s\n' "$NEXT_PUBLIC_APP_URL" > "$ROOT/.tmp/app-url"
setsid pnpm exec next dev --webpack --hostname 127.0.0.1 --port "$PORT" > "$ROOT/.tmp/next.log" 2>&1 &
APP_PID=$!
wait "$APP_PID"
